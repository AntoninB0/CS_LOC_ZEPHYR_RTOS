/* ═══════════════════════════════════════════════════════════════════════════
 * 1) APPLICATION FAMILY — uncomment EXACTLY ONE.
 * 2) CS_NUM_BEACONS — number of reflectors (1 to 5). ALL the timing and sizing
 *    are derived automatically (interval = N x slot, buffers, scheduler);
 *    BUILD_ASSERT guards in cs_config.c.
 *
 * prj.conf overlays per family (SDC reservations, independent of N):
 *   DRONE_FAST_OUTDOOR : none (base prj.conf: CS_EVENT 5000 / SPACING 15000)
 *   BOAT_DOCKING / DRONE_INDOOR, N>=2 : overlay-precision-multi.conf
 *   BOAT_DOCKING / DRONE_INDOOR, N==1 : overlay-precision-single.conf
 * Reminder: reflash the reflectors too if CS_EVENT_LEN changes with the family.
 * Parameter reference: doc_optimised_parameters.ods (Vol 6 Part H cross-refs).
 * ═══════════════════════════════════════════════════════════════════════════ */
#define CONF_DRONE_FAST_OUTDOOR   // fast outdoor drone
//#define CONF_BOAT_DOCKING   // boat docking, short-range precision
//#define CONF_DRONE_INDOOR     // slow indoor drone, dense multipath

#define CS_NUM_BEACONS 1          /* 1..5 */

/* Peer ranging-data retrieval mode: 1 = real-time RAS (fetch ~0, no retention),
 * 0 = on-demand (validated fallback). See iteration 3. */
#define CS_RAS_REALTIME 1

/* ─────────────────────────────────────────────────────────────────────────
 * FAMILY: FAST OUTDOOR DRONE
 * Priority: rate and robustness to motion. Few channels (minimal airtime and
 * data), 15 ms slot per link, RTT sounding (reliable ambiguity resolution
 * beyond 50 m). NOTE on SDC capabilities: 3C/X, 2M_2BT, sounding and NADM
 * documented in the ODS are NOT supported by the SoftDevice Controller (Create
 * Config 0x11 observed) — see the CS_RTT_TYPE block.
 * ───────────────────────────────────────────────────────────────────────── */
#if defined(CONF_DRONE_FAST_OUTDOOR)
#define CS_CHANNEL_THINNING     3     /* ~24 channels, non-ambiguous PBR 50 m */
#define CS_SUBEVENT_LEN_US      4500
#define CS_PROC_LEN_UNITS       40    /* 25 ms: short window, easy to place */
#define CS_MODE0_STEPS          2
#define CS_MAIN_STEPS_MAX       6
#define CS_MODE                 BT_CONN_LE_CS_MAIN_MODE_2_NO_SUB_MODE /* RTT A/B: no submode = pure PBR */
/* ⚠ SDC CAPABILITIES (nrfxlib table, verified on target — else Create Config
 * 0x11): sounding sequence, CSA #3C, 2M_2BT and NADM are NOT supported by the
 * SoftDevice Controller. The AVAILABLE RTT timestamping lever is the random
 * payload (32/64/96/128 bits, supported): finer correlation than AA-only, same
 * role as sounding per the spec (Vol 6 Part H, section 2). Validated bench
 * fallback: BT_CONN_LE_CS_RTT_TYPE_AA_ONLY. */
#define CS_RTT_TYPE             BT_CONN_LE_CS_RTT_TYPE_AA_ONLY
#define CS_SYNC_PHY             BT_CONN_LE_CS_SYNC_2M_PHY      /* 2M_2BT not supported by SDC */
#define CS_CHSEL_TYPE           BT_CONN_LE_CS_CHSEL_TYPE_3B    /* #3C not supported by SDC */
#define CS_CH3C_SHAPE           BT_CONN_LE_CS_CH3C_SHAPE_HAT   /* no effect in 3B */
#define CS_CHMAP_REPETITION     1
#define CS_MAX_TX_POWER         BT_HCI_OP_LE_CS_MAX_MAX_TX_POWER
#define CS_SPACING_UNITS        12    /* 15 ms/link = ACL 3750 + CS 5000 + margin */
#define CS_SINGLE_FLOOR_UNITS   12
/* RAM: at 4-5 beacons, reduced buffers (24 channels → ~30 real steps). */
#define CS_MAX_STEPS_PER_PROC   (CS_NUM_BEACONS >= 4 ? 48 : 64)
#define CS_SCHED_DEFAULT_REPEAT 1

/* ─────────────────────────────────────────────────────────────────────────
 * FAMILY: BOAT DOCKING (precision, < 30 m, slow dynamics)
 * All channels = max resolution; TX 0 dBm (anti-saturation up close); slow and
 * near target → channel_map_repetition 2 when RAM allows (N<=2, see ODS note),
 * otherwise averaging by the scheduler (2 measurements/round). Pure PBR
 * (non-ambiguous 150 m >> envelope) → no RTT, chsel 3B (static).
 * ⚠ N>=4: thinning 2 forced (75 m non-ambiguous, RAM ~/2 per beacon).
 * ───────────────────────────────────────────────────────────────────────── */
#elif defined(CONF_BOAT_DOCKING)
#define CS_CHANNEL_THINNING     (CS_NUM_BEACONS >= 3 ? 2 : 1) // consider a live chmap hot-swap, e.g. via an external controller; check whether it requires restarting the procedure
#define CS_SUBEVENT_LEN_US      8000
#define CS_PROC_LEN_UNITS       100   /* 62.5 ms: ~72 channels to drain */
#define CS_MODE0_STEPS          3
#define CS_MAIN_STEPS_MAX       10
#define CS_MODE                 BT_CONN_LE_CS_MAIN_MODE_2_NO_SUB_MODE
#define CS_RTT_TYPE             BT_CONN_LE_CS_RTT_TYPE_AA_ONLY /* not relevant in pure PBR */
#define CS_SYNC_PHY             BT_CONN_LE_CS_SYNC_2M_PHY
#define CS_CHSEL_TYPE           BT_CONN_LE_CS_CHSEL_TYPE_3B
#define CS_CH3C_SHAPE           BT_CONN_LE_CS_CH3C_SHAPE_HAT   /* unused in 3B */
#define CS_CHMAP_REPETITION     (CS_NUM_BEACONS <= 2 ? 2 : 1)
/* Max TX power (EIRP) of ALL CS transmissions. CEILING: the controller takes the
 * closest supported hardware level <= this value. HCI range -127..+20 dBm;
 * nRF54L15 real max +8 dBm (above that = capped).
 * BT_HCI_OP_LE_CS_MAX_MAX_TX_POWER (=20) = "take the hardware maximum".
 * ⚠ Lowering TX does NOT reduce multipath (direct and reflections attenuated by
 *   the same ratio): the 0 dBm below is short-range anti-saturation (< a few m,
 *   too strong a signal distorts the phase), not a multipath filter.
 *   At range / outdoors, RAISE the TX for SNR.
 * Indicative steps (dBm):
 *      0    tight docking < ~3 m (anti-saturation)
 *     +4    indoor / a few meters
 *     +8    outdoor, nRF54L15 max range
 *   BT_HCI_OP_LE_CS_MAX_MAX_TX_POWER   lets the controller take its max */
#define CS_MAX_TX_POWER         8    /* dBm */
#define CS_SPACING_UNITS        16    /* 20 ms/link = ACL 3750 + CS 8500 + margin */
#define CS_SINGLE_FLOOR_UNITS   16
#define CS_MAX_STEPS_PER_PROC   (CS_NUM_BEACONS <= 2 ? 176 : \
                                 (CS_NUM_BEACONS == 3 ? 96 : 64))
#define CS_SCHED_DEFAULT_REPEAT 2

/* ─────────────────────────────────────────────────────────────────────────
 * FAMILY: SLOW INDOOR DRONE (dense multipath)
 * Frequency diversity is THE remedy to multipath: all channels, reinforced
 * mode-0, max TX (NLOS budget). Slow target → 3B is enough; smoothing comes
 * from the scheduler (2 measurements/round) + gamma weighting on the Python
 * side.
 * ⚠ N>=4: thinning 2 forced (RAM), to avoid if the multipath is severe.
 * ───────────────────────────────────────────────────────────────────────── */
#elif defined(CONF_DRONE_INDOOR)
#define CS_CHANNEL_THINNING     (CS_NUM_BEACONS >= 4 ? 2 : 1)
#define CS_SUBEVENT_LEN_US      8000
#define CS_PROC_LEN_UNITS       100
#define CS_MODE0_STEPS          3
#define CS_MAIN_STEPS_MAX       10
#define CS_MODE                 BT_CONN_LE_CS_MAIN_MODE_2_NO_SUB_MODE
#define CS_RTT_TYPE             BT_CONN_LE_CS_RTT_TYPE_AA_ONLY
#define CS_SYNC_PHY             BT_CONN_LE_CS_SYNC_2M_PHY
#define CS_CHSEL_TYPE           BT_CONN_LE_CS_CHSEL_TYPE_3B
#define CS_CH3C_SHAPE           BT_CONN_LE_CS_CH3C_SHAPE_HAT
#define CS_CHMAP_REPETITION     1
#define CS_MAX_TX_POWER         BT_HCI_OP_LE_CS_MAX_MAX_TX_POWER
#define CS_SPACING_UNITS        16
#define CS_SINGLE_FLOOR_UNITS   16
#define CS_MAX_STEPS_PER_PROC   (CS_NUM_BEACONS >= 4 ? 64 : 96)
#define CS_SCHED_DEFAULT_REPEAT 2

#else
#error "cs_config.h: select exactly one CONF_* family"
#endif

/* ── Common derived values (do not edit) ────────────────────────────────────
 * Connection interval: N==1 → single floor (the SPACING grid only applies
 * between central connections); N>=2 → N x slot, the SDC spaces the ACL anchors
 * by SPACING (deterministic, collision-free).
 * Startup ≈ 11 x interval (LL_CS_REQ/RSP/IND + SDC margin, incompressible in
 * number of events), paid ONCE at arming in free-running
 * (max_procedure_count = 0). The steady-state rate is then
 * procedure_interval x connection_interval per beacon (proportional to N). */
#define CS_MAX_BEACONS          CS_NUM_BEACONS
#define CS_CONN_INTERVAL_UNITS  (CS_NUM_BEACONS == 1 ? CS_SINGLE_FLOOR_UNITS \
                                 : CS_NUM_BEACONS * CS_SPACING_UNITS)
#define CS_ACL_DESYNC_STEP      0  /* precession pointless with the grid;
                                    * fallback 6 if 0x3 aborts get stuck */

# ── Base Bluetooth ──
CONFIG_BT=y
CONFIG_BT_CENTRAL=y
CONFIG_BT_DEVICE_NAME="CS Initiator"
# Must be >= MAX_BEACONS (beacon.h)
CONFIG_BT_MAX_CONN=5
# CRITICAL: even in non-bonding (BT_BONDABLE=n), each encrypted connection
# occupies a slot of the bt_keys pool (key_pool[CONFIG_BT_MAX_PAIRED]) for the
# whole session — the session LTK lives there. Zephyr default = 1: from the 2nd
# beacon on, bt_conn_set_security -> bt_keys_get_addr -> pool full (the single
# slot belongs to the connected beacon, so "in use", impossible to purge) ->
# -ENOMEM (-12) in a loop. parnav_cs had this line.
CONFIG_BT_MAX_PAIRED=5

# ── Security (BT_SECURITY_L2, Just Works) ──
CONFIG_BT_SMP=y
# No bonding: without CONFIG_BT_SETTINGS the keys do not survive a reboot, and
# stale RAM bonds break re-securing (err 2 PIN_OR_KEY_MISSING) and saturate the
# key pool (-ENOMEM). CS security only requires an encrypted link: per-session
# pairing, no storage.
CONFIG_BT_BONDABLE=n

# ── NCS scan (UUID filter + auto connect) ──
CONFIG_BT_SCAN=y
CONFIG_BT_SCAN_FILTER_ENABLE=y
CONFIG_BT_SCAN_UUID_CNT=1

# ── GATT client + service discovery (RAS) ──
CONFIG_BT_GATT_CLIENT=y
CONFIG_BT_GATT_DM=y

# ── Channel Sounding ──
CONFIG_BT_CHANNEL_SOUNDING=y
# One CS context per connection at the controller level: without it (default =
# 1), the CS setup of the 2nd reflector fails for lack of resources
CONFIG_BT_CTLR_SDC_CS_COUNT=5
# Duration RESERVED by the SDC in the timeline for each CS event (default =
# 5000 us). Must be >= max_subevent_len (4500 us in cs_config.c), otherwise
# HCI 0x12 at Set Procedure Parameters.
# ⚠ This is a scheduling RESERVATION: 16000 inflated the footprint of each CS
# slot -> the SDC could not find a placement below ~50-60 ms interval (0x0c)
# and deferred the 1st subevent (~700 ms startup). 5000 = 4500 + margin, sticks
# to the real airtime (~10 ms of useful measurement).
CONFIG_BT_CTLR_SDC_CS_EVENT_LEN_DEFAULT=5000

# ── Compaction of the central scheduling (THE ~700 ms bottleneck) ──
# When BT_CTLR_CHANNEL_SOUNDING=y, the SDC by DEFAULT spaces the ACL anchors of
# the central connections by 90000 us (90 ms!) to fit a CS event behind each
# ACL. With 2 links at 60/77.5 ms interval, this 90 ms grid is untenable: the
# SDC defers the 1st CS subevent until it finds a slot (~10-13 connection
# events) and refuses short intervals. We resize it to our real footprints:
#   ACL (3750) + CS event (5000) + margin ≈ 15000 us per link.
# 2 links x 15 ms = 30 ms -> connection interval 24 (30 ms).
# ⚠ TESTED 10000/interval 16: systematic 0x3 abort on both beacons — 1250 us of
#   margin per slot is insufficient (ACL retransmissions, slippage). 15000
#   validated: startup 340 ms, fetch 130-180 ms, 0 abort. Possible intermediate
#   step to validate: 12500 + interval 20 (25 ms).
CONFIG_BT_CTLR_SDC_CENTRAL_ACL_EVENT_SPACING_DEFAULT=15000
# ACL reservation per link: 7500 us by default, brought down to 3750. The event
# extension (CONFIG_BT_CTLR_SDC_CONN_EVENT_EXTEND_DEFAULT=y by default)
# opportunistically lengthens the event when data remains (RAS fetch) and there
# is no conflict: GATT throughput does not collapse.
CONFIG_BT_CTLR_SDC_MAX_CONN_EVENT_LEN_DEFAULT=3750
# One host reassembly buffer per active CS connection (key = conn_handle, freed
# only at disconnection or at the end of a procedure). Default = 1: a residual
# buffer from an aborted procedure on one link can starve the other link. Align
# on MAX_BEACONS (beacon.h) — CRITICAL in free-running (max_procedure_count=0):
# the N links produce procedures in parallel permanently. 5 = CONFIG_BT_MAX_CONN
# (covers all profiles).
CONFIG_BT_CHANNEL_SOUNDING_REASSEMBLY_BUFFER_CNT=5
# Lets CS and ACL use different PHYs (present in the validated single-beacon
# sample)
CONFIG_BT_TRANSMIT_POWER_CONTROL=y

# ── Ranging Service: client (Ranging REQuestor) ──
CONFIG_BT_RAS=y
CONFIG_BT_RAS_RREQ=y

# ── MTU/ACL: fast transfer of the ranging data ──
CONFIG_BT_L2CAP_TX_MTU=247
CONFIG_BT_BUF_ACL_TX_SIZE=251
CONFIG_BT_BUF_ACL_RX_SIZE=251
CONFIG_BT_ATT_PREPARE_COUNT=2

# ── Floats (atan2f, regression) ──
CONFIG_FPU=y
# Several threads use the FPU (ranging in main + display_thread which logs
# floats): without FPU_SHARING the FPU registers are corrupted on a context
# switch
CONFIG_FPU_SHARING=y
# Formatting of the %f in LOG_INF (otherwise corrupted output)
CONFIG_CBPRINTF_FP_SUPPORT=y

# ── Misc ──
CONFIG_DK_LIBRARY=y
CONFIG_SERIAL=y
CONFIG_LOG=y
CONFIG_MAIN_STACK_SIZE=4096
CONFIG_SYSTEM_WORKQUEUE_STACK_SIZE=4096

# ── RTT: enlarged buffer so the boot logs are not lost before the JLink
#    terminal attaches (default = 1 KB, skip mode) ──
CONFIG_USE_SEGGER_RTT=y
CONFIG_LOG_BACKEND_RTT=y
CONFIG_SEGGER_RTT_BUFFER_SIZE_UP=4096
CONFIG_UART_INTERRUPT_DRIVEN=y
